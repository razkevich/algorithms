// Values-UP template int best = Integer.MIN_VALUE; public int maxPathSum(TreeNode root) { oneArm(root); return best; } // Returns the best oneArm reach down through this node (so the parent can extend). // Updates `best` with the bothArms value (full path turning at this node). int oneArm(TreeNode n) { if (n == null) return 0; int L = Math.max(oneArm(n.left), 0); // clamp negatives (skip that side) int R = Math.max(oneArm(n.right), 0); best = Math.max(best, n.val + L + R); // bothArms: update global (side effect) return n.val + Math.max(L, R); // oneArm: return upward (pick ONE side) }
